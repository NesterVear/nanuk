# Nanuk — stack de desarrollo (repos oficiales)

# Editor y TUIs
neovim
lazygit
lazydocker
git-delta          # diffs bonitos (lazygit/git lo usan)
github-cli

# Contenedores
docker
docker-compose
docker-buildx

# Lenguajes que van a nivel de sistema
python
python-pip
uv
php
composer
nodejs
npm
lua
luarocks
tree-sitter-cli

# Gestores de versiones (para tener varias versiones de cada lenguaje)
mise               # node / python / go / ruby / java...  → `nanuk lang`
rustup             # toolchains de Rust                   → `nanuk lang rust`
